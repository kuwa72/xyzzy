# Alternative build system for xyzzy.
#
# The canonical build is xyzzy.sln (MSVC).  This CMake build exists so the same
# sources can be compiled with mingw-w64 and exercised under Wine, which makes
# build+test possible from a macOS or Linux host and in CI.  See tools/x.
cmake_minimum_required(VERSION 3.20)

project(xyzzy CXX RC)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)

set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/src)
set(GEN ${CMAKE_CURRENT_BINARY_DIR}/gen)
file(MAKE_DIRECTORY ${GEN})

# The sources are Shift_JIS (CP932): a second byte of 0x5C ('\') inside a
# double-byte character would otherwise be taken for an escape and break the
# string literal.  The execution charset stays CP932 because xyzzy's internal
# character handling is CP932-based.
#
# src/ is added with -iquote, not -I: it holds string.h, list.h, search.h etc.,
# which would shadow the CRT headers on an -I path.  MSVC resolves quoted
# includes relative to the including file, and the vcxproj files deliberately
# keep src/ off the include path; -iquote reproduces that.
set(XYZZY_LANG_OPTIONS
  -finput-charset=CP932
  -fexec-charset=CP932
  -iquote ${SRC}
  -include ${CMAKE_CURRENT_SOURCE_DIR}/compat/mingw/xyzzy-compat.h
  # The sources predate the alternative operator tokens and use "or", "compl"
  # and friends as ordinary identifiers, which MSVC still permits.
  -fno-operator-names
  # Braced initialisers mixing an unsigned enum with an int field: an error
  # under C++11 rules, accepted by the MSVC dialect this code was written for.
  -Wno-narrowing)

# Wine resolves a plain absolute POSIX path against the current drive, so every
# path handed to a cross-built generator is passed relative to its working
# directory instead.
file(RELATIVE_PATH GEN_FROM_SRC ${SRC} ${GEN})

# generate_source(<generator> <output> [ARGS ...] [DEPENDS ...])
#   Runs the generator with stdout captured into <output>.
function(generate_source tool output)
  cmake_parse_arguments(GS "" "" "ARGS;DEPENDS" ${ARGN})
  # The generators are built for the target, so they run through the emulator.
  set(command ${CMAKE_CROSSCOMPILING_EMULATOR} $<TARGET_FILE:${tool}> ${GS_ARGS})
  add_custom_command(
    OUTPUT ${GEN}/${output}
    COMMAND ${CMAKE_COMMAND}
            "-DXY_CMD=${command}"
            -DXY_OUT=${GEN}/${output}
            -DXY_WD=${SRC}
            -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/run-capture.cmake
    DEPENDS ${tool} ${GS_DEPENDS} ${CMAKE_CURRENT_SOURCE_DIR}/cmake/run-capture.cmake
    WORKING_DIRECTORY ${SRC}
    COMMENT "Generating gen/${output}"
    VERBATIM)
endfunction()

# Headers generated into ${GEN} are reached both as "foo.h" and as "gen/foo.h"
# (src/symtable.cc includes "gen/symtable.cc").
target_include_directories(xyzzy PRIVATE
  ${SRC}/privctrl
  ${GEN}
  ${CMAKE_CURRENT_BINARY_DIR})

target_compile_definitions(xyzzy PRIVATE
  _CRT_SECURE_NO_DEPRECATE
  STRICT
  WIN32
  _WIN32
  _MT
  __XYZZY__
  PRIVCTRL_NODLL
  HAVE_DIMM_H
  $<$<CONFIG:Debug>:DEBUG>
  $<$<CONFIG:Debug>:DEBUG_GC>)

target_compile_options(xyzzy PRIVATE
  ${XYZZY_LANG_OPTIONS}
  -fpermissive
  -fno-strict-aliasing
  -fms-extensions
  -Wno-write-strings
  -Wno-invalid-offsetof
  -Wno-unknown-pragmas)

target_link_libraries(xyzzy PRIVATE
  comctl32 comdlg32 gdi32 imm32 mpr ole32 oleaut32 rpcrt4 shell32 shlwapi
  user32 uuid version winspool wsock32 crypt32 secur32 z
  # src/dll.cc marshals foreign calls with libffi outside the MSVC x86 build

add_dependencies(xyzzy version-describe)

set_target_properties(xyzzy PROPERTIES
  RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})

# ---------------------------------------------------------------------------
# companion executables
# ---------------------------------------------------------------------------

foreach(app xyzzycli xyzzyenv)
  # Unlike xyzzy.rc these are compiled as they are (see the vcxproj files); the
  # script only rewrites the resource path separators.
  add_custom_command(
    OUTPUT ${GEN}/${app}.rc
    COMMAND ${CMAKE_COMMAND}
            "-DXY_IN=${SRC}/${app}.rc"
            -DXY_OUT=${GEN}/${app}.rc
            -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/merge-rc.cmake
    DEPENDS ${SRC}/${app}.rc
            ${CMAKE_CURRENT_SOURCE_DIR}/cmake/merge-rc.cmake
    COMMENT "Generating gen/${app}.rc"
    VERBATIM)
  add_executable(${app} WIN32 ${SRC}/${app}.cc ${GEN}/${app}.rc)
  target_include_directories(${app} PRIVATE ${GEN})
  target_compile_definitions(${app} PRIVATE
    _CRT_SECURE_NO_DEPRECATE STRICT WIN32 _WIN32 _MT)
  target_compile_options(${app} PRIVATE
    ${XYZZY_LANG_OPTIONS} -fpermissive -fno-strict-aliasing -fms-extensions)
  target_link_libraries(${app} PRIVATE user32 shell32 advapi32)
  set_target_properties(${app} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
endforeach()
